cmake_minimum_required(VERSION 3.16)
project(RigStartupGuard LANGUAGES CXX)
add_executable(guard_probe guard_probe.cpp)
target_compile_features(guard_probe PRIVATE cxx_std_11)
add_executable(guard_contract_test guard_contract_test.cpp)
target_compile_features(guard_contract_test PRIVATE cxx_std_11)
if(MSVC)
    target_compile_options(guard_probe PRIVATE /W4 /WX)
    target_compile_options(guard_contract_test PRIVATE /W4 /WX)
else()
    target_compile_options(guard_probe PRIVATE -Wall -Wextra -Werror)
    target_compile_options(guard_contract_test PRIVATE -Wall -Wextra -Werror)
endif()
